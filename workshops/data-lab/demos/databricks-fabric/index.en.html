Eliminates conflicting margin definitions across SQL, Genie, and BI by locking official KPI logic in Unity Catalog: 

version: 0.1
source: zava_retail.gold.fact_store_daily_summary
dimensions:
  - name: region_name
    expr: region_name
  - name: fulfillment_hub
    expr: hub_name
measures:
  - name: net_margin_rate
    expr: (SUM(gross_sales_cny) - SUM(cogs_cny) - SUM(spoilage_cost_cny) - SUM(promo_discount_cny)) / NULLIF(SUM(gross_sales_cny), 0)
  - name: stockout_rate
    expr: SUM(stockout_sku_days) * 1.0 / NULLIF(SUM(scheduled_sku_days), 0)
