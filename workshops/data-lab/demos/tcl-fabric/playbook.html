| project AlertTimestamp, ShipmentId, VesselCode, OriginPort, DestinationPort, AffectedProductId, AffectedUnits, CargoValueUSD, DelayDays, PortCongestionIndex

-- 3. Lakehouse SQL Analytics Endpoint — Cross-Domain SAP Master + Zero-Copy EDI 852 Polarization Check
SELECT d.DistributorId, d.DistributorName, d.Country, i.ProductId,
       i.OnHandUnits, i.WeeksOfSupply, i.PriceProtectionExposureUSD, i.StockHealthStatus
FROM dbo.fact_channel_inventory_weekly i
JOIN dbo.dim_distributors d ON i.DistributorId = d.DistributorId
WHERE i.IsLatestSnapshot = 1
  AND (i.WeeksOfSupply < 1.5 OR i.PriceProtectionExposureUSD > 50000)
ORDER BY i.WeeksOfSupply ASC;
