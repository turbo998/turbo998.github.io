  - name: region_name
    expr: region_name

  - name: category
    expr: category
measures:
  - name: total_sales_cny
    expr: SUM(sales_amount_cny)
  - name: net_margin_rate
    expr: SUM(net_margin_cny) / NULLIF(SUM(sales_amount_cny), 0)
    comment: "唯一受控净毛利率公式，供 AI/BI Genie 与 Power BI 共用"

CREATE FUNCTION zava_retail.silver.mask_phone(phone STRING)
