Copy-Paste Implementation Template

// Core dual-channel service pattern: turnEpoch prevents stale tool results after an interruption
let turnEpoch = 0;

ws.on("message", async (raw) => {
  const evt = JSON.parse(raw);

  // 1) Customer interruption: increment turnEpoch and cancel current speech output
  if (evt.type === "input_audio_buffer.speech_started") {
    turnEpoch += 1;
    ws.send(JSON.stringify({ type: "response.cancel" }));
  }

  // 2) The model requests an order lookup or refund calculation
  if (evt.type === "response.function_call_arguments.done") {
    const myEpoch = turnEpoch;
    const result = await executeDeterministicCrmTool(evt.name, JSON.parse(evt.arguments));
    if (myEpoch !== turnEpoch) return; // Discard stale results if the customer interrupted and began a new turn

    ws.send(JSON.stringify({
      type: "conversation.item.create",
      item: { type: "function_call_output", call_id: evt.call_id, output: JSON.stringify(result) }
    }));
    ws.send(JSON.stringify({ type: "response.create" }));
  }
});
